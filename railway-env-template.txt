# ============ APPLICATION ============
APP_NAME="Paper Haven"
APP_ENV=production
APP_DEBUG=false
APP_KEY=base64:YOUR_KEY_HERE_GENERATE_WITH_php_artisan_key:generate
APP_TIMEZONE=UTC
APP_URL=${RAILWAY_PUBLIC_DOMAIN}

# ============ DATABASE ============
# These use Railway's MySQL service variables
# Make sure you've added MySQL database service first!

DB_HOST=${MYSQLHOST}
DB_PORT=${MYSQLPORT}
DB_DATABASE=${MYSQLDATABASE}

DB_PASSWORD=${MYSQLPASSWORD}

# ============ SESSION & CACHE ============

SESSION_LIFETIME=120
SESSION_ENCRYPT=false
SESSION_PATH=/

CACHE_PREFIX=

# ============ LOGGING ============
LOG_CHANNEL=errorlog
LOG_LEVEL=error

# ============ QUEUE ============
QUEUE_CONNECTION=database

# ============ MAIL (Optional) ============
MAIL_MAILER=log

MAIL_PORT=2525
MAIL_USERNAME=null
MAIL_PASSWORD=null

MAIL_FROM_NAME="${APP_NAME}"

# ============ FILE SYSTEM ============
FILESYSTEM_DISK=local

# ==========================================
# IMPORTANT NOTES:
# ==========================================
# 1. Generate APP_KEY locally first:
#    Run: php artisan key:generate --show
#    Copy the output (including "base64:")
#
# 2. MySQL Variables:
#    ${MYSQLHOST}, ${MYSQLPORT}, etc. are automatically
#    provided by Railway when you add a MySQL service
#
# 3. Domain:
#    ${RAILWAY_PUBLIC_DOMAIN} is provided by Railway
#    after you generate a domain for your service
#
# 4. Don't use quotes around Railway variable references:
#    ✅ Correct: DB_HOST=${MYSQLHOST}
#    ❌ Wrong: DB_HOST="${MYSQLHOST}"
